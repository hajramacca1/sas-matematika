Kode Google Apps Script

Buat Google Sheet baru → Ekstensi → Apps Script → tempel kode ini → Terapkan (Deploy) sebagai Aplikasi Web (Akses: Siapa saja / Anyone):

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    result.push({
      waktu: rows[i][0] ? Utilities.formatDate(new Date(rows[i][0]), "GMT+8", "dd/MM/yyyy HH:mm:ss") : "-",
      nama: rows[i][1],
      kelas: rows[i][2],
      token: rows[i][3],
      skor: rows[i][4],
      benar: rows[i][5],
      pelanggaran: rows[i][6],
      logPelanggaran: rows[i][7],
      status: rows[i][8]
    });
  }
  return ContentService.createTextOutput(JSON.stringify(result.reverse()))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = (e && e.postData && e.postData.contents) ? JSON.parse(e.postData.contents) : {
      nama: "Tes Manual Editor", kelas: "9.1", token: "Matematika9", skor: 100, 
      benar: 30, pelanggaran: 0, logPelanggaran: "Tes Run Berhasil", status: "Simulasi"
    };
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Waktu", "Nama Murid", "Kelas", "Token", "Skor", "Benar", "Pelanggaran", "Detail Pelanggaran", "Status"]);
    }
    sheet.appendRow([
      new Date(), data.nama, data.kelas, data.token, data.skor, 
      data.benar, data.pelanggaran, data.logPelanggaran, data.status
    ]);
    return ContentService.createTextOutput(JSON.stringify({"result":"success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
